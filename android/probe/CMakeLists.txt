# RR64 RT64 probe: builds RT64 (+ plume) for arm64-v8a and links it into
# the SDLActivity-loaded "main" library. No game code, no N64Recomp, no ROM.
#
# Configure inputs (set by scripts/build_probe.py via Gradle arguments):
#   RR64_RT64_DIR        - repo native/lib/rt64 (fetched by
#                          scripts/setup_dependencies.py, Android-patched by
#                          scripts/build_probe.py)
#   RR64_SDL_DIR         - build-probe/deps/SDL (fetched by build_probe.py)
#   RR64_HOST_TOOLS_DIR  - build-probe/host/bin (host-built file_to_c)
#   RR64_HOST_DXC_BIN    - host dxc binary (x86_64 dxc-linux / dxc.exe)
#   RR64_HOST_DXC_LIB    - host dxc lib dir (for LD_LIBRARY_PATH on Linux)
#
# The Android patches in android/probe/patches/ must already be applied to
# the rt64/plume trees by scripts/build_probe.py before this runs.
#
# NOT compile-checked: no NDK / Android Studio on this machine.
cmake_minimum_required(VERSION 3.20)
project(rr64probe LANGUAGES C CXX)

if(NOT DEFINED RR64_RT64_DIR)
    message(FATAL_ERROR "RR64_RT64_DIR is not set (run scripts/build_probe.py)")
endif()
if(NOT DEFINED RR64_SDL_DIR)
    message(FATAL_ERROR "RR64_SDL_DIR is not set (run scripts/build_probe.py)")
endif()
if(NOT DEFINED RR64_HOST_TOOLS_DIR)
    message(FATAL_ERROR "RR64_HOST_TOOLS_DIR is not set (run scripts/build_probe.py)")
endif()
if(NOT DEFINED RR64_HOST_DXC_BIN OR NOT DEFINED RR64_HOST_DXC_LIB)
    message(FATAL_ERROR "RR64_HOST_DXC_BIN / RR64_HOST_DXC_LIB are not set (run scripts/build_probe.py)")
endif()
if(NOT ANDROID)
    message(FATAL_ERROR "The probe only builds for Android (NDK toolchain)")
endif()
if(CMAKE_HOST_WIN32)
    set(CMAKE_HOST_EXECUTABLE_SUFFIX ".exe")
endif()

set(RR64_RT64_DIR "${RR64_RT64_DIR}")
set(RR64_SDL_DIR "${RR64_SDL_DIR}")
if(NOT EXISTS "${RR64_RT64_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "RT64 tree missing at ${RR64_RT64_DIR}: run scripts/setup_dependencies.py, then scripts/build_probe.py")
endif()
if(NOT EXISTS "${RR64_SDL_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "SDL tree missing at ${RR64_SDL_DIR}: run scripts/build_probe.py")
endif()

# --- SDL2, built from source for arm64-v8a (shared, as the game uses it). ---
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${RR64_SDL_DIR}" "${CMAKE_BINARY_DIR}/SDL")
set(SDL2_INCLUDE_DIRS "${RR64_SDL_DIR}/include")
set(SDL2_LIBRARIES SDL2::SDL2)
# Lets RT64's find_package(SDL2) use the SDL we just built instead of
# looking for a system package (same trick as AeroGauge's FindSDL2.cmake).
list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")

# --- Host file_to_c: shader binaries are generated by a HOST executable,
# never an Android one. ---
if(NOT EXISTS "${RR64_HOST_TOOLS_DIR}/file_to_c${CMAKE_HOST_EXECUTABLE_SUFFIX}")
    message(FATAL_ERROR "Host file_to_c missing at ${RR64_HOST_TOOLS_DIR}: run scripts/build_probe.py")
endif()
add_executable(file_to_c IMPORTED GLOBAL)
set_target_properties(file_to_c PROPERTIES IMPORTED_LOCATION
    "${RR64_HOST_TOOLS_DIR}/file_to_c${CMAKE_HOST_EXECUTABLE_SUFFIX}")

# --- Host DXC: HLSL -> SPIR-V runs on the BUILD machine (x86_64). RT64's
# patched CMakeLists picks this up as ${RR64_HOST_DXC} on ANDROID. ---
if(CMAKE_HOST_WIN32)
    set(RR64_HOST_DXC "${RR64_HOST_DXC_BIN}")
else()
    set(RR64_HOST_DXC ${CMAKE_COMMAND} -E env
        "LD_LIBRARY_PATH=${RR64_HOST_DXC_LIB}"
        "${RR64_HOST_DXC_BIN}")
endif()

# --- RT64 (static) + plume, at the RR64 pins. ---
set(RT64_STATIC TRUE CACHE BOOL "" FORCE)
set(RT64_SDL_WINDOW_VULKAN TRUE CACHE BOOL "" FORCE)
set(PLUME_SDL_VULKAN_ENABLED ON CACHE BOOL "" FORCE)
# 16 KiB page size: required for Android 15+ (same as AeroGauge).
add_link_options("-Wl,-z,max-page-size=16384")
add_subdirectory("${RR64_RT64_DIR}" "${CMAKE_BINARY_DIR}/rt64")

# --- Probe library. SDLActivity loads "main" and calls SDL_main. ---
# NOTE: SDL2main is deliberately NOT linked: on Android, SDL's
# SDLActivity calls SDL_main directly. If the link fails with an
# undefined SDL_main reference, add SDL2main back here.
add_library(main SHARED src/main/cpp/probe.cpp)
# RT64's own include dirs are directory-scoped inside its tree, so the
# probe target needs the same list explicitly (from RT64's CMakeLists).
target_include_directories(main PRIVATE
    "${RR64_RT64_DIR}/src"
    "${RR64_RT64_DIR}/src/contrib"
    "${RR64_RT64_DIR}/src/contrib/plume"
    "${RR64_RT64_DIR}/src/contrib/imgui"
    "${RR64_RT64_DIR}/src/contrib/hlslpp/include"
    "${RR64_RT64_DIR}/src/contrib/mupen64plus-core/src/api"
    "${RR64_RT64_DIR}/src/contrib/nativefiledialog-extended/src/include"
)
target_link_libraries(main PRIVATE rt64 plume SDL2::SDL2 android log)
